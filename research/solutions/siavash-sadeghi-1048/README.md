```sh
python -m pip install -r requirements.txt
python fresh_audit_diagnostics.py > diagnostic_rerun.json
python -O fresh_audit_diagnostics.py > diagnostic_rerun_optimized.json
python -m unittest test_diagnostics.py
python -O -m unittest test_diagnostics.py
pdflatex -no-shell-escape -interaction=nonstopmode -halt-on-error research_nonoverlap_proof.tex
pdflatex -no-shell-escape -interaction=nonstopmode -halt-on-error research_nonoverlap_proof.tex
pdflatex -no-shell-escape -interaction=nonstopmode -halt-on-error review_note.tex
pdflatex -no-shell-escape -interaction=nonstopmode -halt-on-error review_note.tex
```

Run `python scripts/catalogue.py --check --catalogue-only` from the repository root. NumPy is needed only for supplementary diagnostics.
